The stat Parameter

model$posterior$C_pm$plot_density(ylim = c(0., 1.))

f = function(x) {
  mu = mean(x)
  sigma = stats::sd(x)
  return(list(
    density = \(...) return(\(x) return(stats::dnorm(x, mu, sigma, FALSE))),
    quantile = \(q, ...) return(stats::qnorm(q, mu, sigma, TRUE, FALSE)),
    mean = \(...) return(mu),
    median = \(...) return(mu),
    sd = \(...) return(sigma)
  ))
}

model$update(stat = f)
model$run()
